## Design

**Addresses** are `uint32_t` in host byte order. `ip_parse()` and `cidr_parse()` share one
hand-written scanner: each octet is 1 to 3 digits, no leading zero (except a lone `0`),
at most 255, with exactly three dots and nothing left over. The prefix is parsed the same
way with a maximum of 32. Nothing is written to the output unless the whole string is valid.

**The mask and /0.** `cidr_mask()` returns 0 for a prefix of 0 before it shifts, so the
undefined `<< 32` never happens, and clamps anything above 32 to 32. Everything else is
built from the mask: network is `addr & mask`, broadcast is `addr | ~mask`, and
`cidr_contains()` compares `addr & mask` with the block's `addr & mask`. Because the
default route's mask is 0, it contains every address with no special case.

**Host counts.** `cidr_host_count()` works in 64 bits so a /0 gives 2^32 - 2. A /31
(RFC 3021) has two usable addresses and its first and last host are the network and
broadcast addresses; a /32 has one, and all of network, broadcast, first and last host
are that address. /0 to /30 use network + 1 and broadcast - 1.

**Subnets.** `cidr_subnet()` rejects a non-network block, an `n` that is zero or not a
power of two, a new prefix over 32, and `i >= n`. The new prefix is the old one plus
log2(n), and subnet `i` starts at `network + i * 2^(32 - new_prefix)`, computed in 64 bits.
It returns one subnet per call, so `myapp split 0.0.0.0/0 2147483648` streams its output
instead of building an array.

**The table.** A `fwd_table_t` is a growable array of `route_t` (capacity starts at 4 and
doubles with `realloc`). `table_add()` copies the route and refuses host bits or a second
route for the same address and prefix. `table_load()` reads with `getline()`, strips
comments, skips blank lines, and stops at the first bad line, reporting its number; the
line buffer is freed on every path.

**Longest prefix match.** `table_lookup()` scans the whole table and keeps the matching
route with the largest prefix, so the result does not depend on insertion order. Since a
block can only appear once, no two matches have the same prefix. A trie would make lookup
cost at most 32 steps regardless of table size, but a linear scan is simple and enough here.

**Command line.** Exit codes are 0 for success, 1 for a bad command line or malformed
address, 2 for an unreadable table or a bad line (the table is loaded before any address
is looked at), and 3 when an address has no route. A lookup run reports every address,
then exits 1 if any was malformed, otherwise 3 if any was unreachable.

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

static bool is_digit(char c)
{
  return c >= '0' && c <= '9';
}

/**
 * Parse a run of decimal digits with no leading zero (a lone "0" is fine) and a
 * value no larger than max. Returns the first character after the digits, or
 * NULL if there is no number or it is not acceptable.
 */
static const char *parse_number(const char *s, uint32_t max, uint32_t *value)
{
  if (!is_digit(*s))
  {
    return NULL;
  }
  if (s[0] == '0' && is_digit(s[1]))
  {
    return NULL;
  }

  uint32_t v = 0;
  while (is_digit(*s))
  {
    v = v * 10 + (uint32_t)(*s - '0');
    if (v > max)
    {
      return NULL;
    }
    s++;
  }
  *value = v;
  return s;
}

/** Parse a dotted quad at the start of s; returns the character after it or NULL. */
static const char *parse_quad(const char *s, uint32_t *addr)
{
  uint32_t result = 0;
  for (int i = 0; i < 4; i++)
  {
    uint32_t octet;
    s = parse_number(s, 255, &octet);
    if (s == NULL)
    {
      return NULL;
    }
    result = (result << 8) | octet;
    if (i < 3)
    {
      if (*s != '.')
      {
        return NULL;
      }
      s++;
    }
  }
  *addr = result;
  return s;
}

bool ip_parse(const char *text, uint32_t *addr)
{
  if (text == NULL)
  {
    return false;
  }
  uint32_t value;
  const char *end = parse_quad(text, &value);
  if (end == NULL || *end != '\0')
  {
    return false;
  }
  if (addr != NULL)
  {
    *addr = value;
  }
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  snprintf(out, IP_STR_LEN, "%u.%u.%u.%u", (addr >> 24) & 0xFFu, (addr >> 16) & 0xFFu,
           (addr >> 8) & 0xFFu, addr & 0xFFu);
}

bool cidr_parse(const char *text, cidr_t *block)
{
  if (text == NULL)
  {
    return false;
  }
  uint32_t addr;
  const char *end = parse_quad(text, &addr);
  if (end == NULL || *end != '/')
  {
    return false;
  }
  uint32_t prefix;
  end = parse_number(end + 1, 32, &prefix);
  if (end == NULL || *end != '\0')
  {
    return false;
  }
  if (block != NULL)
  {
    block->addr = addr;
    block->prefix = (uint8_t)prefix;
  }
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN])
{
  char ip[IP_STR_LEN];
  ip_format(block.addr, ip);
  snprintf(out, CIDR_STR_LEN, "%s/%u", ip, block.prefix > 32 ? 32u : (unsigned)block.prefix);
}

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

uint32_t cidr_mask(uint8_t prefix)
{
  if (prefix == 0)
  {
    return 0;
  }
  if (prefix > 32)
  {
    prefix = 32;
  }
  return 0xFFFFFFFFu << (32 - prefix);
}

uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
  return block.addr | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block)
{
  uint32_t network = cidr_network(block);
  return block.prefix >= 31 ? network : network + 1;
}

uint32_t cidr_last_host(cidr_t block)
{
  uint32_t broadcast = cidr_broadcast(block);
  return block.prefix >= 31 ? broadcast : broadcast - 1;
}

uint64_t cidr_host_count(cidr_t block)
{
  if (block.prefix >= 32)
  {
    return 1;
  }
  if (block.prefix == 31)
  {
    return 2;
  }
  return (UINT64_C(1) << (32 - block.prefix)) - 2;
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block)
{
  return block.addr == cidr_network(block);
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  if (out == NULL || !cidr_is_network(block) || n == 0 || (n & (n - 1)) != 0)
  {
    return false;
  }

  uint32_t bits = 0;
  while ((UINT32_C(1) << bits) < n)
  {
    bits++;
  }

  uint32_t prefix = block.prefix > 32 ? 32 : block.prefix;
  uint32_t new_prefix = prefix + bits;
  if (new_prefix > 32 || i >= n)
  {
    return false;
  }

  uint64_t offset = (uint64_t)i << (32 - new_prefix);
  out->addr = (uint32_t)(cidr_network(block) + offset);
  out->prefix = (uint8_t)new_prefix;
  return true;
}

/** Longest field route_parse() needs to look at, with room for its '\0'. */
#define FIELD_LEN 32

/** Copy the next whitespace separated field into buf; false if none or too long. */
static bool next_field(const char **p, char buf[FIELD_LEN])
{
  const char *s = *p + strspn(*p, WHITESPACE);
  size_t len = strcspn(s, WHITESPACE);
  if (len == 0 || len >= FIELD_LEN)
  {
    return false;
  }
  memcpy(buf, s, len);
  buf[len] = '\0';
  *p = s + len;
  return true;
}

bool route_parse(const char *line, route_t *route)
{
  if (line == NULL)
  {
    return false;
  }

  char dest[FIELD_LEN];
  char gateway[FIELD_LEN];
  char dev[FIELD_LEN];
  const char *p = line;
  if (!next_field(&p, dest) || !next_field(&p, gateway) || !next_field(&p, dev))
  {
    return false;
  }
  if (p[strspn(p, WHITESPACE)] != '\0' || strlen(dev) >= DEV_NAME_LEN)
  {
    return false;
  }

  route_t result = {0};
  if (strcmp(dest, "default") == 0)
  {
    result.dest.addr = 0;
    result.dest.prefix = 0;
  }
  else if (!cidr_parse(dest, &result.dest) || !cidr_is_network(result.dest))
  {
    return false;
  }

  if (strcmp(gateway, "direct") == 0)
  {
    result.gateway = 0;
  }
  else if (!ip_parse(gateway, &result.gateway))
  {
    return false;
  }

  strcpy(result.dev, dev);
  if (route != NULL)
  {
    *route = result;
  }
  return true;
}

struct fwd_table
{
  route_t *routes;
  size_t count;
  size_t capacity;
};

fwd_table_t *table_create(void)
{
  return calloc(1, sizeof(fwd_table_t));
}

void table_destroy(fwd_table_t *table)
{
  if (table == NULL)
  {
    return;
  }
  free(table->routes);
  free(table);
}

size_t table_count(const fwd_table_t *table)
{
  return table == NULL ? 0 : table->count;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
  if (table == NULL || route == NULL || !cidr_is_network(route->dest))
  {
    return false;
  }
  for (size_t i = 0; i < table->count; i++)
  {
    if (table->routes[i].dest.addr == route->dest.addr &&
        table->routes[i].dest.prefix == route->dest.prefix)
    {
      return false;
    }
  }

  if (table->count == table->capacity)
  {
    size_t capacity = table->capacity == 0 ? 4 : table->capacity * 2;
    route_t *grown = realloc(table->routes, capacity * sizeof(route_t));
    if (grown == NULL) // GCOVR_EXCL_START
    {
      return false;
    } // GCOVR_EXCL_STOP
    table->routes = grown;
    table->capacity = capacity;
  }

  table->routes[table->count++] = *route;
  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
  if (bad_line != NULL)
  {
    *bad_line = 0;
  }
  if (table == NULL || in == NULL)
  {
    return false;
  }

  char *line = NULL;
  size_t cap = 0;
  size_t number = 0;
  bool ok = true;

  while (getline(&line, &cap, in) != -1)
  {
    number++;
    char *comment = strchr(line, '#');
    if (comment != NULL)
    {
      *comment = '\0';
    }
    if (line[strspn(line, WHITESPACE)] == '\0')
    {
      continue;
    }

    route_t route;
    if (!route_parse(line, &route) || !table_add(table, &route))
    {
      if (bad_line != NULL)
      {
        *bad_line = number;
      }
      ok = false;
      break;
    }
  }

  free(line);
  return ok && !ferror(in);
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
  if (table == NULL)
  {
    return NULL;
  }
  const route_t *best = NULL;
  for (size_t i = 0; i < table->count; i++)
  {
    const route_t *r = &table->routes[i];
    if (cidr_contains(r->dest, dest) && (best == NULL || r->dest.prefix > best->dest.prefix))
    {
      best = r;
    }
  }
  return best;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H

#include "lab.h"
#include <errno.h>
#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#ifdef TEST
#define main main_exclude
#endif

enum
{
  EXIT_OK = 0,
  EXIT_USAGE = 1,
  EXIT_TABLE = 2,
  EXIT_NO_ROUTE = 3
};

static void print_usage(FILE *out)
{
  fprintf(out, "Usage: myapp info <address>/<prefix>\n"
               "       myapp split <network>/<prefix> <count>\n"
               "       myapp lookup <table-file> <address>...\n");
}

static int usage_error(const char *message)
{
  fprintf(stderr, "myapp: %s\n", message);
  print_usage(stderr);
  return EXIT_USAGE;
}

static int cmd_info(const char *text)
{
  cidr_t block;
  if (!cidr_parse(text, &block))
  {
    fprintf(stderr, "myapp: %s: not a valid address/prefix\n", text);
    return EXIT_USAGE;
  }

  char buf[IP_STR_LEN];
  ip_format(block.addr, buf);
  printf("%-11s%s\n", "address", buf);
  printf("%-11s/%u\n", "prefix", (unsigned)block.prefix);
  ip_format(cidr_mask(block.prefix), buf);
  printf("%-11s%s\n", "netmask", buf);
  ip_format(cidr_network(block), buf);
  printf("%-11s%s\n", "network", buf);
  ip_format(cidr_broadcast(block), buf);
  printf("%-11s%s\n", "broadcast", buf);
  ip_format(cidr_first_host(block), buf);
  printf("%-11s%s\n", "first host", buf);
  ip_format(cidr_last_host(block), buf);
  printf("%-11s%s\n", "last host", buf);
  printf("%-11s%" PRIu64 "\n", "hosts", cidr_host_count(block));
  return EXIT_OK;
}

static int cmd_split(const char *text, const char *count_text)
{
  cidr_t block;
  if (!cidr_parse(text, &block))
  {
    fprintf(stderr, "myapp: %s: not a valid network/prefix\n", text);
    return EXIT_USAGE;
  }
  if (!cidr_is_network(block))
  {
    fprintf(stderr, "myapp: %s: has host bits set, so it is not a network\n", text);
    return EXIT_USAGE;
  }

  char *end;
  errno = 0;
  unsigned long count = strtoul(count_text, &end, 10);
  if (count_text[0] < '0' || count_text[0] > '9' || *end != '\0' || errno != 0 ||
      count > UINT32_MAX)
  {
    fprintf(stderr, "myapp: %s: not a valid count\n", count_text);
    return EXIT_USAGE;
  }

  uint32_t n = (uint32_t)count;
  cidr_t subnet;
  if (!cidr_subnet(block, n, 0, &subnet))
  {
    fprintf(stderr, "myapp: cannot split %s into %s subnets: the count must be a power of two "
                    "that leaves a prefix of at most /32\n",
            text, count_text);
    return EXIT_USAGE;
  }

  for (uint32_t i = 0; i < n; i++)
  {
    char buf[CIDR_STR_LEN];
    cidr_subnet(block, n, i, &subnet);
    cidr_format(subnet, buf);
    puts(buf);
  }
  return EXIT_OK;
}

static int cmd_lookup(const char *path, int count, char **addresses)
{
  FILE *in = fopen(path, "r");
  if (in == NULL)
  {
    fprintf(stderr, "myapp: %s: %s\n", path, strerror(errno));
    return EXIT_TABLE;
  }

  fwd_table_t *table = table_create();
  if (table == NULL) // GCOVR_EXCL_START
  {
    fclose(in);
    fprintf(stderr, "myapp: out of memory\n");
    return EXIT_TABLE;
  } // GCOVR_EXCL_STOP

  size_t bad_line = 0;
  bool loaded = table_load(table, in, &bad_line);
  fclose(in);
  if (!loaded)
  {
    if (bad_line != 0)
    {
      fprintf(stderr, "myapp: %s:%zu: not a valid route\n", path, bad_line);
    }
    else
    {
      fprintf(stderr, "myapp: %s: could not be read\n", path);
    }
    table_destroy(table);
    return EXIT_TABLE;
  }

  bool malformed = false;
  bool unreachable = false;
  for (int i = 0; i < count; i++)
  {
    uint32_t addr;
    if (!ip_parse(addresses[i], &addr))
    {
      fprintf(stderr, "myapp: %s: not a valid address\n", addresses[i]);
      malformed = true;
      continue;
    }

    const route_t *route = table_lookup(table, addr);
    if (route == NULL)
    {
      printf("%s unreachable\n", addresses[i]);
      unreachable = true;
    }
    else if (route->gateway != 0)
    {
      char gateway[IP_STR_LEN];
      ip_format(route->gateway, gateway);
      printf("%s via %s dev %s\n", addresses[i], gateway, route->dev);
    }
    else
    {
      printf("%s dev %s\n", addresses[i], route->dev);
    }
  }

  table_destroy(table);
  if (malformed)
  {
    return EXIT_USAGE;
  }
  return unreachable ? EXIT_NO_ROUTE : EXIT_OK;
}

int main(int argc, char **argv)
{
  if (argc == 1)
  {
    print_usage(stdout);
    return EXIT_OK;
  }

  const char *command = argv[1];
  if (strcmp(command, "info") == 0 && argc == 3)
  {
    return cmd_info(argv[2]);
  }
  if (strcmp(command, "split") == 0 && argc == 4)
  {
    return cmd_split(argv[2], argv[3]);
  }
  if (strcmp(command, "lookup") == 0 && argc >= 4)
  {
    return cmd_lookup(argv[2], argc - 3, argv + 3);
  }
  return usage_error("bad command line");
}

void test_subnet(void)
{
  const char *expected[] = {"192.168.1.0/26", "192.168.1.64/26", "192.168.1.128/26",
                            "192.168.1.192/26"};
  for (uint32_t i = 0; i < 4; i++)
  {
    cidr_t out;
    char buf[CIDR_STR_LEN];
    TEST_ASSERT_TRUE(cidr_subnet(block("192.168.1.0/24"), 4, i, &out));
    cidr_format(out, buf);
    TEST_ASSERT_EQUAL_STRING(expected[i], buf);
  }

  cidr_t out;
  TEST_ASSERT_TRUE(cidr_subnet(block("10.0.0.0/8"), 1, 0, &out));
  TEST_ASSERT_EQUAL_UINT8(8, out.prefix);
  TEST_ASSERT_EQUAL_HEX32(0x0A000000, out.addr);

  /* 0.0.0.0/0 into 2^31 pieces: the last one is 255.255.255.254/31. */
  TEST_ASSERT_TRUE(cidr_subnet(block("0.0.0.0/0"), UINT32_C(1) << 31, (UINT32_C(1) << 31) - 1, &out));
  TEST_ASSERT_EQUAL_HEX32(0xFFFFFFFE, out.addr);
  TEST_ASSERT_EQUAL_UINT8(31, out.prefix);

  TEST_ASSERT_TRUE(cidr_subnet(block("0.0.0.0/0"), 2, 1, &out));
  TEST_ASSERT_EQUAL_HEX32(0x80000000, out.addr);
  TEST_ASSERT_EQUAL_UINT8(1, out.prefix);

  /* Splitting down to single hosts. */
  TEST_ASSERT_TRUE(cidr_subnet(block("10.0.0.0/30"), 4, 3, &out));
  TEST_ASSERT_EQUAL_HEX32(0x0A000003, out.addr);
  TEST_ASSERT_EQUAL_UINT8(32, out.prefix);
}

void test_table_add(void)
{
  fwd_table_t *t = table_create();
  route_t r = make_route("10.0.0.0/8", "10.1.0.1", "eth1");
  TEST_ASSERT_TRUE(table_add(t, &r));
  TEST_ASSERT_EQUAL_UINT(1, table_count(t));

  /* Duplicate block, even with different gateway/device. */
  route_t dup = make_route("10.0.0.0/8", "direct", "eth9");
  TEST_ASSERT_FALSE(table_add(t, &dup));
  /* Same address, different prefix is a different block. */
  route_t other = make_route("10.0.0.0/16", "direct", "eth9");
  TEST_ASSERT_TRUE(table_add(t, &other));
  /* Same prefix, different address is a different block. */
  route_t other2 = make_route("11.0.0.0/8", "direct", "eth9");
  TEST_ASSERT_TRUE(table_add(t, &other2));
  TEST_ASSERT_EQUAL_UINT(3, table_count(t));

  route_t hostbits = {{0x0A000001, 8}, 0, "eth0"};
  TEST_ASSERT_FALSE(table_add(t, &hostbits));
  TEST_ASSERT_FALSE(table_add(t, NULL));
  TEST_ASSERT_FALSE(table_add(NULL, &r));
  TEST_ASSERT_EQUAL_UINT(3, table_count(t));

  /* The table keeps a copy. */
  strcpy(r.dev, "changed");
  TEST_ASSERT_EQUAL_STRING("eth1", table_lookup(t, ip("10.200.0.1"))->dev);
  table_destroy(t);
}

void test_lookup_independent_of_order(void)
{
  const char *lines[] = {"10.0.0.0/8 10.1.0.1 eth1",   "10.1.0.0/16 direct eth1",
                         "10.1.2.0/24 direct eth2",    "10.1.2.128/25 10.1.2.1 eth2",
                         "0.0.0.0/0 9.9.9.9 eth0"};
  const size_t n = sizeof lines / sizeof lines[0];
  const char *probes[] = {"10.1.2.200", "10.1.2.5", "10.1.9.9", "10.9.9.9", "8.8.8.8"};
  const uint8_t want[] = {25, 24, 16, 8, 0};

  fwd_table_t *fwd = table_create();
  fwd_table_t *rev = table_create();
  for (size_t i = 0; i < n; i++)
  {
    route_t a, b;
    TEST_ASSERT_TRUE(route_parse(lines[i], &a));
    TEST_ASSERT_TRUE(route_parse(lines[n - 1 - i], &b));
    TEST_ASSERT_TRUE(table_add(fwd, &a));
    TEST_ASSERT_TRUE(table_add(rev, &b));
  }
  for (size_t i = 0; i < sizeof probes / sizeof probes[0]; i++)
  {
    const route_t *a = table_lookup(fwd, ip(probes[i]));
    const route_t *b = table_lookup(rev, ip(probes[i]));
    TEST_ASSERT_NOT_NULL(a);
    TEST_ASSERT_NOT_NULL(b);
    TEST_ASSERT_EQUAL_UINT8(want[i], a->dest.prefix);
    TEST_ASSERT_EQUAL_UINT8(want[i], b->dest.prefix);
    TEST_ASSERT_EQUAL_HEX32(a->dest.addr, b->dest.addr);
  }
  table_destroy(fwd);
  table_destroy(rev);
}
